cmake_minimum_required(VERSION 3.20)
project(marching_cubes VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MC_ENABLE_SANITIZERS "Build with address and undefined behavior sanitizers" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SILO REQUIRED IMPORTED_TARGET silo)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale)
find_package(PNG REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
add_library(mc_core src/extract.cpp src/silo_reader.cpp src/render.cpp src/frame.cpp src/media.cpp src/storage.cpp src/movie.cpp src/export.cpp)
target_include_directories(mc_core PUBLIC include)
target_include_directories(mc_core SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(mc_core PUBLIC PkgConfig::SILO PNG::PNG PkgConfig::AV nlohmann_json::nlohmann_json ${OpenCV_LIBS})
target_compile_options(mc_core PRIVATE -Wall -Wextra -Wpedantic)
if(MC_ENABLE_SANITIZERS)
    target_compile_options(mc_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mc_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(marching_cubes src/main.cpp)
target_link_libraries(marching_cubes PRIVATE mc_core)
target_compile_options(marching_cubes PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    foreach(component extract silo render media storage)
        add_executable(test_${component} tests/test_${component}.cpp)
        target_link_libraries(test_${component} PRIVATE mc_core)
        target_compile_options(test_${component} PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
        add_test(NAME ${component} COMMAND test_${component})
    endforeach()
    add_executable(make_pipeline_fixture tests/make_fixture.cpp)
    target_link_libraries(make_pipeline_fixture PRIVATE PkgConfig::SILO)
    target_compile_options(make_pipeline_fixture PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(test_frame tests/test_frame.cpp)
    target_link_libraries(test_frame PRIVATE mc_core)
    target_compile_options(test_frame PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    add_test(NAME frame COMMAND test_frame $<TARGET_FILE:make_pipeline_fixture>)
    add_executable(test_pipeline tests/test_pipeline.cpp)
    target_link_libraries(test_pipeline PRIVATE nlohmann_json::nlohmann_json)
    target_compile_options(test_pipeline PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    add_test(NAME pipeline COMMAND test_pipeline $<TARGET_FILE:marching_cubes> $<TARGET_FILE:make_pipeline_fixture>)
    set_tests_properties(pipeline PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180)
endif()
install(TARGETS marching_cubes RUNTIME DESTINATION bin)
